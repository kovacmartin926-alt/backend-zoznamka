async function poslat() { const input = document.getElementById('userInput'); const window = document.getElementById('chat-window'); // TENTO RIADOK JE DÔLEŽITÝ - musí brať aktuálnu hodnotu pri každom kliknutí! const postava = document.getElementById('postava').value; const text = input.value; if (!text) return; window.innerHTML += `
>> TY: ${text}
`; input.value = ""; window.scrollTop = window.scrollHeight; try { const res = await fetch('/poslat_spravu', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ postava: postava, // Tu sa posiela vybratá postava text: text }) }); const data = await res.json(); // Zobrazenie odpovede so správnym menom window.innerHTML += `
${postava.toUpperCase()}: ${data.odpoved}
`; } catch (e) { window.innerHTML += `
!!! ERROR !!!
`; } window.scrollTop = window.scrollHeight; }
